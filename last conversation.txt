import sys
from pathlib import Path
sys.path.append(str(Path('.').resolve()))
from config.settings import DATA_DIR, TRADING_CONFIG
import pandas as pd

pairs = TRADING_CONFIG['pairs']
tfs = ['1min', '5min', '15min', '1h', '4h', '1d']
thresholds = {'1min': 200000, '5min': 50000, '15min': 20000, '1h': 90000, '4h': 20000, '1d': 3600}

partial = []
missing = []
for symbol in pairs:
    for tf in tfs:
        path = DATA_DIR / f'{symbol}_{tf}_history.parquet'
        if not path.exists():
            missing.append((symbol, tf))
            continue
        try:
            df = pd.read_parquet(path)
        except:
            missing.append((symbol, tf))
            continue
        if len(df) < thresholds[tf]:
            partial.append((symbol, tf, len(df), thresholds[tf]))

print('MISSING HISTORY ({}):'.format(len(missing)))
for m in missing: print('  ', m)
print()
print('PARTIAL HISTORY ({}):'.format(len(partial)))
for p in sorted(partial, key=lambda x: x[2]/x[3]): print('  ', p[0], p[1], '- has', p[2], '/ needs', p[3], '({:.1f}%)'.format(100*p[2]/p[3]))

from pathlib import Path

from ai_engine.training.model_status import get_model_status

status = get_model_status()
# Check coverage per symbol - which symbols have no active model
by_symbol = {}
for row in status:
    sym = row['symbol']
    tf = row['timeframe']
    if sym not in by_symbol:
        by_symbol[sym] = {'timeframes': [], 'weak_timeframes': [], 'no_model': [], 'weak_signal_count': 0}
    if tf in ('1min','5min','15min'):  # the used timeframes for trading
        by_symbol[sym]['timeframes'].append(tf)
        if row['active_model_version'] is None:
            by_symbol[sym]['no_model'].append(tf)
        if row.get('weak_signal'):
            by_symbol[sym]['weak_signal_count'] += 1
            by_symbol[sym]['weak_timeframes'].append(tf)

print('=== SYMBOL COVERAGE (ACTIVE TRADE TIMEFRAMES) ===')
for sym in sorted(by_symbol.keys()):
    s = by_symbol[sym]
    flag = ''
    if s['no_model']:
        flag += ' NO_MODEL:' + ','.join(s['no_model'])
    if s['weak_signal_count'] > 0:
        flag += ' WEAK@' + ','.join(s['weak_timeframes'])
    print(f'  {sym}: {len([\"x\" for t in s[\"timeframes\"] if t not in s[\"no_model\"]])}/3 models active {flag}')

In settings.py:87-88 :

- Weights include market_structure: 0.10 — but in the entire codebase, no SignalSource.MARKET_STRUCTURE signal is ever added !

Once a trade goes +10 pips, we should move SL to break-even to guarantee no loss on pullbacks. Currently we only support:

- SL hit → close
